\section{Functional Requirements}\label{sec:fr}
\begin{guidance}
Write each functional requirement in User Story format with a confirmation part. Every confirmation must be testable. Group stories by module. ``Traces to'' links each story to a user need (UN) or risk (R).
\end{guidance}

\end{table}

\subsection{Module 1: User Account Management}

\begin{userstory}{FR-01}{User registration}{High}{UN-01}
  \storyas{new user}
  \storywant{to register with my phone number and a password}
  \storyso{I can access the system's features}
  \begin{confirmation}
    \item Confirm that a user can register with a valid phone number and a password of at least 8 characters.
    \item Confirm that registration with an already-registered phone number shows an error.
    \item Confirm that an OTP is sent and the account is activated only after it is verified.
  \end{confirmation}
\end{userstory}

\begin{userstory}{FR-02}{User login}{High}{UN-01, R-02}
  \storyas{registered user}
  \storywant{to log in with my credentials}
  \storyso{I can use my account securely}
  \begin{confirmation}
    \item Confirm that valid credentials log the user in and open the dashboard.
    \item Confirm that invalid credentials show a generic error message.
    \item Confirm that the account is locked for 15 minutes after 5 consecutive failed attempts.
  \end{confirmation}
\end{userstory}
