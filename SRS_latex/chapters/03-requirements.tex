\section{Functional Requirements}\label{sec:fr}
\begin{guidance}
Write each functional requirement in User Story format with a confirmation part. Every confirmation must be testable. Group stories by module. ``Traces to'' links each story to a user need (UN) or risk (R).
\end{guidance}

\subsection{Module 1: User Account Management}

\begin{userstory}{FR-01}{User registration}{High}{UN-01}
  \storyas{new user}
  \storywant{to register with my phone number and a password}
  \storyso{I can access the system's features}
  \begin{confirmation}
    \item Confirm that a user can register with a valid phone number and a password of at least 8 characters.
    \item Confirm that registration with an already-registered phone number shows an error.
    \item Confirm that an OTP is sent and the account is activated only after it is verified.
  \end{confirmation}
\end{userstory}

\begin{userstory}{FR-02}{User login}{High}{UN-01, R-02}
  \storyas{registered user}
  \storywant{to log in with my credentials}
  \storyso{I can use my account securely}
  \begin{confirmation}
    \item Confirm that valid credentials log the user in and open the dashboard.
    \item Confirm that invalid credentials show a generic error message.
    \item Confirm that the account is locked for 15 minutes after 5 consecutive failed attempts.
  \end{confirmation}
\end{userstory}

\subsection{Parent Module}
The parent has read-only access to the data of the linked children.

\begin{userstory}{FR-P1}{Login and Dashboard}{High}{UN-P1, R-02, R-03}
  \storyas{parent}
  \storywant{to log in and see a dashboard for my child (or children)}
  \storyso{I can quickly see how they are doing}
  \begin{confirmation}
    \item Confirm that the parent logs in with email and password (the common login rules apply).
    \item Confirm that if more than one child is linked, the parent can switch between children from the dashboard.
    \item Confirm that the dashboard shows each child's attendance rate, pending assignments, latest grade, outstanding fees and recent notifications.
    \item Confirm that a parent with no linked child sees a message to contact the center.
  \end{confirmation}
\end{userstory}

\begin{userstory}{FR-P2}{View Child Progress}{High}{UN-P2, R-03}
  \storyas{parent}
  \storywant{to see my child's marks, level progress and teacher feedback}
  \storyso{I can understand how my child is performing in each subject}
  \begin{confirmation}
    \item Confirm that the parent can view results per subject and per assessment only after the teacher has published them. Unpublished grades show ``Unpublished''.
    \item Confirm that the parent can see the grade breakdown by category (Homework, Quizzes, Exams) and the overall progress bar.
    \item Confirm that the parent can read the teacher's feedback on assignments and results.
    \item Confirm that the parent can see a summary of overall performance, such as average marks and completed versus missed tasks.
  \end{confirmation}
\end{userstory}

\begin{userstory}{FR-P3}{View Child Attendance}{High}{UN-P3, R-05}
  \storyas{parent}
  \storywant{to see my child's attendance record}
  \storyso{I know if my child is missing classes}
  \begin{confirmation}
    \item Confirm that the parent can view attendance by date, course and status (Present, Absent, Late, Excused).
    \item Confirm that the parent can see the attendance percentage and filter by date range.
    \item Confirm that low attendance is highlighted visibly.
    \item Confirm that attendance is read-only for the parent.
  \end{confirmation}
\end{userstory}

\begin{userstory}{FR-P4}{Track Assignment Status}{Medium}{UN-P3}
  \storyas{parent}
  \storywant{to see my child's assignments and their status}
  \storyso{I can make sure homework is completed on time}
  \begin{confirmation}
    \item Confirm that the parent sees each assignment's course, title, due date, marks and status (Pending, Submitted, Overdue).
    \item Confirm that overdue assignments are displayed in a separate, highlighted area, and the parent receives a notification when an assignment becomes overdue.
    \item Confirm that draft assignments are never visible.
    \item Confirm that the parent can view but cannot submit or edit anything for the child.
  \end{confirmation}
\end{userstory}

\begin{userstory}{FR-P5}{View Fee Invoices}{High}{UN-P4, R-04}
  \storyas{parent}
  \storywant{to see my child's fee invoices}
  \storyso{I know how much to pay and by when}
  \begin{confirmation}
    \item Confirm that the parent can see each invoice with invoice ID, course, billing period, total fee, discount, amount payable and due date.
    \item Confirm that the parent can view the invoice status (Paid, Partially Paid, Due, Overdue) and download an invoice as a PDF.
    \item Confirm that the parent can see only the invoices of his or her own linked children.
  \end{confirmation}
\end{userstory}

\begin{userstory}{FR-P6}{View Payment Records}{Medium}{UN-P4, R-04}
  \storyas{parent}
  \storywant{to see my payment history and receipts}
  \storyso{I can verify what I have paid and what is still due}
  \begin{confirmation}
    \item Confirm that the parent can see payment date, amount, method, transaction ID and remaining balance for each payment.
    \item Confirm that partial payments are shown with the updated outstanding balance.
    \item Confirm that the parent can download a receipt for any recorded payment.
    \item Confirm that the parent cannot edit or delete payment records.
  \end{confirmation}
\end{userstory}

\begin{userstory}{FR-P7}{Receive Notifications}{Medium}{UN-P5, R-10}
  \storyas{parent}
  \storywant{to get notifications about my child's activities}
  \storyso{I do not miss important updates}
  \begin{confirmation}
    \item Confirm that the parent is notified of absence, overdue assignment, newly published grade or feedback, new invoice, upcoming due date, payment recorded and teacher or center announcements.
    \item Confirm that the parent can see a notification list, mark items as read and filter by type.
    \item Confirm that the parent can choose which notification types to receive (in-app or email).
  \end{confirmation}
\end{userstory}
